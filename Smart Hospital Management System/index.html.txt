function distanceKm(a,b,c,d){
 const r=6371,toRad=x=>x*Math.PI/180;
 const x=toRad(c-a),y=toRad(d-b);
 const h=Math.sin(x/2)**2+Math.cos(toRad(a))*Math.cos(toRad(c))*Math.sin(y/2)**2;
 return 2*r*Math.asin(Math.sqrt(h));
}
async function findNearestHospitals(){
 const status=$("#nearbyStatus"),results=$("#hospitalResults");
 if(!navigator.geolocation){status.textContent="Location is not supported. Use Google Maps instead.";mapSearch("hospitals near me");return}
 status.textContent="Requesting your location...";
 navigator.geolocation.getCurrentPosition(async pos=>{
  const lat=pos.coords.latitude,lon=pos.coords.longitude;
  status.textContent="Searching nearby hospital listings...";
  try{
   const query=`[out:json][timeout:20];(node["amenity"="hospital"](around:15000,${lat},${lon});way["amenity"="hospital"](around:15000,${lat},${lon});relation["amenity"="hospital"](around:15000,${lat},${lon}););out center tags;`;
   const res=await fetch("https://overpass-api.de/api/interpreter",{method:"POST",headers:{"Content-Type":"text/plain;charset=UTF-8"},body:query});
   if(!res.ok)throw Error("Hospital search unavailable");
   const data=await res.json();
   const arr=(data.elements||[]).map(e=>{
    const t=e.tags||{};
    return {name:t.name||t["name:en"]||"Hospital",lat:e.lat??e.center?.lat,lon:e.lon??e.center?.lon,address:[t["addr:street"],t["addr:suburb"],t["addr:city"]].filter(Boolean).join(", ")};
   }).filter(h=>Number.isFinite(h.lat)&&Number.isFinite(h.lon));
   const unique=[...new Map(arr.map(h=>[h.name+"|"+h.lat+"|"+h.lon,h])).values()]
    .sort((a,b)=>distanceKm(lat,lon,a.lat,a.lon)-distanceKm(lat,lon,b.lat,b.lon)).slice(0,15);
   results.innerHTML=unique.map(h=>`
    <div class="card hospital"><div class="hospital-icon">🏥</div><div>
     <h3>${esc(h.name)}</h3><p>${distanceKm(lat,lon,h.lat,h.lon).toFixed(1)} km away (approx.)</p>
     <p>${esc(h.address||"Nearby location")}</p>
     <button class="btn small" onclick="mapSearch('${esc(h.name)}')">View Map</button>
     <button class="btn light small" onclick="window.open('https://www.google.com/maps/dir/?api=1&origin=${lat},${lon}&destination=${h.lat},${h.lon}','_blank')">Directions</button>
    </div></div>`).join("")||"<div class='card empty'>No nearby hospital records found.</div>";
   status.textContent="Results from OpenStreetMap. Distances are approximate.";
  }catch(e){
   status.textContent="Could not load live listings. Opening Google Maps.";
   mapSearch("hospitals near me");
  }
 },()=>{status.textContent="Location permission denied or unavailable.";mapSearch("hospitals near me")},{enableHighAccuracy:true,timeout:12000});
}
